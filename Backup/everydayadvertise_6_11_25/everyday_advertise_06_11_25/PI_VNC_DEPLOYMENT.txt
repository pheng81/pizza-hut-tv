================================================================================

================================================================================

SSH to Pi:
ssh pi@<PI_IP_ADDRESS>

Install required packages:
pip3 install mss pillow

================================================================================
STEP 3: Integrate VNC Handlers into complete_pi_client.py
================================================================================

Edit the Pi client:
nano complete_pi_client.py

--- ADD AT TOP (after imports) ---
from pi_vnc_tunnel import init_vnc_tunnel, get_vnc_tunnel

--- ADD AFTER socketio.connect() ---
# Initialize VNC tunnel
vnc_tunnel = init_vnc_tunnel(socketio, PI_ID)
logging.info('✅ VNC tunnel initialized')

--- ADD NEW SOCKETIO HANDLERS (after existing handlers) ---

@socketio.on('vnc_connect')
def handle_vnc_connect(data):
    """Server requests VNC connection from dashboard"""
    dashboard_sid = data.get('dashboard_sid')
    pi_id = data.get('pi_id')
    
    logging.info(f'🖥️ VNC connect request from dashboard {dashboard_sid}')
    
    if pi_id != PI_ID:
        logging.warning(f'⚠️ VNC connect for wrong Pi: {pi_id}')
        return
    
    tunnel = get_vnc_tunnel()
    if tunnel:
        success = tunnel.connect(dashboard_sid)
        if not success:
            logging.error('❌ Failed to start VNC tunnel')
    else:
        logging.error('❌ VNC tunnel not initialized')

@socketio.on('vnc_data')
def handle_vnc_data_from_dashboard(data):
    """Receive VNC data from dashboard (mouse/keyboard)"""
    pi_id = data.get('pi_id')
    
    if pi_id != PI_ID:
        return
    
    tunnel = get_vnc_tunnel()
    if tunnel:
        tunnel.send_to_vnc(data)

@socketio.on('vnc_disconnect')
def handle_vnc_disconnect(data):
    """Dashboard disconnects VNC"""
    pi_id = data.get('pi_id')
    dashboard_sid = data.get('dashboard_sid')
    
    logging.info(f'🖥️ VNC disconnect from dashboard {dashboard_sid}')
    
    if pi_id != PI_ID:
        return
    
    tunnel = get_vnc_tunnel()
    if tunnel:
        tunnel.disconnect()

--- SAVE AND EXIT (Ctrl+X, Y, Enter) ---

================================================================================
TROUBLESHOOTING
================================================================================

❌ "ModuleNotFoundError: No module named 'pi_vnc_tunnel'"
→ Make sure pi_vnc_tunnel.py is in same directory as complete_pi_client.py

❌ "ModuleNotFoundError: No module named 'mss'"
→ Install: pip3 install mss

❌ VNC window shows "Connection Failed"
→ Check Pi is connected in dashboard first
→ Check Pi logs: sudo journalctl -u pizzahut-tv-pi -f

❌ Black screen in VNC window
→ Check mss working: python3 -c "import mss; print(mss.mss().monitors)"
→ Check display is on
→ Check DISPLAY environment variable

❌ "Pi not connected" error
→ Make sure Pi client is running
→ Check WebSocket connection in dashboard
